Looking back

Over the last five chapters we built sumcsv v1 cleanly in standard C23, error paths and golden suite included. What changes if a new request asks to sum multiple lines, select a single column, handle quoted fields with escaped commas and quotes, and lift the line-length cap?

A. It is not just a handful of if statements. Once line length and unquoted cell length are unbounded, you need two dynamically grown buffers, each checking capacity multiplication overflow and preserving old storage when realloc fails — plus a cleanup ladder (goto cleanup) so every failing branch frees both buffers cleanly.

★ Plain C can certainly build it to the end, yet keeping five things by hand everywhere — ownership, copying, cleanup, error plumbing and where memory comes from — begins taking half the file.

The need for this chapter, and its context

This chapter closes Part XIII and builds the bridge into Part XIV (chapter 106 onward). Before introducing a library, we push plain standard C through a real v2 expansion of sumcsv so we can measure with our own eyes how far standard C goes cleanly and where manual bookkeeping starts generating friction.

By the end of this chapter

By the end of this chapter you run sumcsv_v2.c — handling multi-line streams, column selection, quoted fields with "" escapes, unbounded dynamic lines and embedded-NUL refusal — and compare v1 against v2 side by side to see the exact problems the API designs in Part XIV aim to move into types and signatures.

105.1 Four new requirements in plain C#

The demonstration below implements all four v2 features in standard C23 alone: multiple rows from a stream, summing only target_col, double-quoted fields with "" unescaping, and dynamically grown line and cell buffers.

examples-en/sumcsv/06-next/sumcsv_v2.c

/* Version 2 grown by hand in plain C: multiline, column selection, quoted fields, and dynamic buffers. */
#include <ctype.h>
#include <errno.h>
#include <inttypes.h>
#include <limits.h>
#include <stdckdint.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct dynbuf {
    char  *data;
    size_t len;
    size_t cap;
};

static bool buf_push(struct dynbuf *b, char ch)
{
    if (b->len + 1 >= b->cap) {
        size_t ncap = 0;
        if (b->cap == 0) {
            ncap = 32;
        } else if (ckd_mul(&ncap, b->cap, (size_t)2)) {
            return false;
        }
        char *nb = realloc(b->data, ncap);
        if (!nb) return false;
        b->data = nb;
        b->cap = ncap;
    }
    b->data[b->len++] = ch;
    b->data[b->len] = '\0';
    return true;
}

static int read_dyn_line(FILE *in, struct dynbuf *b, const char **why)
{
    b->len = 0;
    if (b->data) b->data[0] = '\0';

    int ch = fgetc(in);
    if (ch == EOF) {
        if (ferror(in)) { *why = "read error"; return -1; }
        return 0;
    }
    do {
        if (ch == '\0') { *why = "embedded NUL byte"; return -1; }
        if (!buf_push(b, (char)ch)) { *why = "out of memory"; return -1; }
        if (ch == '\n') break;
        ch = fgetc(in);
    } while (ch != EOF);

    if (ferror(in)) { *why = "read error"; return -1; }
    return 1;
}

static bool parse_i64_exact(const char *s, int64_t *out, const char **why)
{
    while (*s == ' ' || *s == '\t') s++;
    if (*s == '\0') { *why = "empty cell"; return false; }

    const char *digits = s;
    if (*digits == '+' || *digits == '-') digits++;
    if (!isdigit((unsigned char)*digits)) { *why = "bad integer"; return false; }

    errno = 0;
    char *end = nullptr;
    long long v = strtoll(s, &end, 10);
    if (end == s) { *why = "bad integer"; return false; }
    if (errno == ERANGE) { *why = "integer out of range"; return false; }
#if LLONG_MIN < INT64_MIN || LLONG_MAX > INT64_MAX
    if (v < INT64_MIN || v > INT64_MAX) { *why = "integer out of range"; return false; }
#endif
    while (*end == ' ' || *end == '\t') end++;
    if (*end != '\0') { *why = "trailing garbage"; return false; }
    *out = (int64_t)v;
    return true;
}

static bool eval_row_col(const char *line, size_t target_col, struct dynbuf *cell,
                         int64_t *out_val, const char **why)
{
    const char *p = line;
    size_t col = 0;
    bool found = false;

    for (;;) {
        col++;
        cell->len = 0;
        if (cell->data) cell->data[0] = '\0';

        if (*p == '"') {
            p++;
            for (;;) {
                if (*p == '\0' || *p == '\n' || (*p == '\r' && p[1] == '\n')) {
                    *why = "unclosed quote";
                    return false;
                }
                if (*p == '"') {
                    if (p[1] == '"') {
                        if (!buf_push(cell, '"')) { *why = "out of memory"; return false; }
                        p += 2;
                        continue;
                    }
                    p++;
                    break;
                }
                if (!buf_push(cell, *p)) { *why = "out of memory"; return false; }
                p++;
            }
            while (*p == ' ' || *p == '\t') p++;
        } else {
            while (*p && *p != ',' && *p != '\n' && !(*p == '\r' && p[1] == '\n')) {
                if (*p == '"') { *why = "stray quote"; return false; }
                if (!buf_push(cell, *p)) { *why = "out of memory"; return false; }
                p++;
            }
        }

        if (col == target_col) {
            if (!parse_i64_exact(cell->data ? cell->data : "", out_val, why)) return false;
            found = true;
        }

        if (*p == ',') { p++; continue; }
        if (*p == '\r' && p[1] == '\n') p += 2;
        else if (*p == '\n') p++;
        if (*p != '\0') { *why = "unexpected char after field"; return false; }
        break;
    }

    if (!found) {
        *why = "missing target column";
        return false;
    }
    return true;
}

static bool sum_csv_stream(FILE *in, size_t target_col, int64_t *out_sum, const char **out_why)
{
    struct dynbuf line = {0};
    struct dynbuf cell = {0};
    int64_t acc = 0;
    size_t rows = 0;
    bool ok = false;

    for (;;) {
        int rc = read_dyn_line(in, &line, out_why);
        if (rc < 0) goto cleanup;
        if (rc == 0) break;

        rows++;
        int64_t v = 0;
        if (!eval_row_col(line.data, target_col, &cell, &v, out_why)) goto cleanup;
        if (ckd_add(&acc, acc, v)) { *out_why = "sum overflow"; goto cleanup; }
    }

    if (rows == 0) {
        *out_why = "empty input";
        goto cleanup;
    }
    *out_sum = acc;
    ok = true;

cleanup:
    free(line.data);
    free(cell.data);
    return ok;
}

static void run_case(const char *label, size_t col, const char *csv)
{
    FILE *fp = tmpfile();
    if (!fp) {
        printf("  %-18s -> err: tmpfile failed\n", label);
        return;
    }
    size_t n = strlen(csv);
    if (n > 0 && fwrite(csv, 1, n, fp) != n) {
        fclose(fp);
        printf("  %-18s -> err: write failed\n", label);
        return;
    }
    rewind(fp);

    int64_t sum = -1;
    const char *why = "";
    if (sum_csv_stream(fp, col, &sum, &why)) {
        printf("  %-18s col=%zu -> ok: %" PRId64 "\n", label, col, sum);
    } else {
        printf("  %-18s col=%zu -> err: %s (sum untouched: %" PRId64 ")\n",
               label, col, why, sum);
    }
    fclose(fp);
}

int main(void)
{
    run_case("quoted multiline", 2,
             "\"widget, small\",100\n\"bolt \"\"hex\"\"\",250\nplain,-30\r\n");
    run_case("unclosed quote",   2, "\"widget,100\n");
    run_case("missing col",      3, "a,10\n");
    run_case("bad target cell",  2, "\"ok,name\",abc\n");
    return 0;
}

Output

  quoted multiline   col=2 -> ok: 320
  unclosed quote     col=2 -> err: unclosed quote (sum untouched: -1)
  missing col        col=3 -> err: missing target column (sum untouched: -1)
  bad target cell    col=2 -> err: bad integer (sum untouched: -1)

The file has grown to 208 lines, carrying four new guards.

  • struct dynbuf grown with ckd_mul — buf_push checks b->cap * 2 against size_t overflow with ckd_mul before calling realloc, and leaves the existing b->data intact if realloc returns null.
  • Blocking embedded NUL bytes — a '\0' byte inside a CSV stream tricks C string helpers into stopping early, so read_dyn_line rejects ch == '\0' on the spot as "embedded NUL byte".
  • Quote state and "" unescaping — fields opening with " treat inner commas as literal bytes and collapse doubled "" into one " inside the cell buffer, while rejecting "unclosed quote" and mid-field "stray quote".
  • goto cleanup and intact caller output — sum_csv_stream owns two dynamic buffers (line and cell), so every error path (read failure, parse error, empty stream, sum overflow) meets at cleanup: to free both, overwriting *out_sum only on success.

105.2 The friction that gathers between v1 and v2#

Nothing in v1 was broken, yet expanding it to one small 208-line v2 program added manual work across five axes at once.

★ None of these lines is wrong C — all of them compile cleanly under -Werror and pass the sanitisers. The issue is scale: here we have only two dynbufs, yet as soon as rows accumulate into dynamic record arrays or nested fields, buffer growth guards, cleanup: ladders and ownership rules must be repeated in every function, and missing one turns into a leak, silent truncation, or use after free.

A common misconception. “If plain C can build everything anyway, why examine the library contracts in Part XIV?”

That plain C can build it is true — sumcsv_v2.c just proved it in 208 lines. What Part XIV asks is not “can plain C do it”, but how the code changes when those five habits kept in human heads — errors as values, length-carrying views, allocator parameters, failure atomicity, and refusing rather than truncating — move into types and function signatures. Only after writing the plain-C version by hand can you weigh that contrast honestly.

Q. Does sumcsv_v2.c also allow literal newline characters inside a quoted field?

A. No. Because read_dyn_line stops reading a row at the first \n, a literal newline inside quotes is rejected as "unclosed quote". Supporting RFC 4180 multi-line quoted cells requires making the line reader itself quote-aware across newlines, turning the stream reader into a multi-row state machine.

Q. dynbuf works fine here — why list hardcoded realloc as a drawback?

A. For a one-shot CLI process that reads stdin and exits, calling the heap directly is completely reasonable. In an environment where dynamic heap allocation is banned (chapter 99), or a server resetting a request arena all at once, baking realloc into buf_push makes sum_csv_stream unusable without rewriting its allocation path.
