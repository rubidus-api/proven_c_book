돌아보기

101장 와 102장 로 한 줄의 구문과 합산 넘침은 전부 잡았다. 그러면 main 에서 fgets(line, sizeof line, stdin) 으로 한 줄을 읽어 sum_line 에 넘기고 printf("%lld\n", sum) 하면 도구가 완성되는가?

답. 아직 바깥 스트림의 경계 네 군데가 뚫려 있다. 첫째, fgets 가 널을 돌렸을 때 「빈 입력(EOF)」과 「장치 읽기 오류(ferror)」를 가리지 않으면 디스크 오류가 빈 파일처럼 보인다. 둘째, 버퍼보다 긴 줄은 오류가 아니라 잘린 앞토막만 들어오고, 뒷토막이 그대로 버려진다.

★ 셋째, 1판 계약은 한 줄만 받기로 했는데 둘째 줄이 와도 아무도 읽지 않으면 그냥 무시된다. 넷째, 디스크가 차거나 파이프가 끊겨 fprintf 나 fflush 에서 쓰기가 실패해도 종료 코드 0 으로 끝날 수 있다(78장).

이 장의 필요성과 맥락

문자열 위에서는 파싱이 끝나지만, 명령줄 도구는 stdin 과 stdout 사이에서 산다. 78장 에서 따로 배운 fgets 잘림 판별, ferror 구분, 버퍼 비우기 실패 검사가 sumcsv 라는 한 프로그램의 입구와 출구에서 어떻게 맞물리는지 이 장에서 끝까지 확인한다.

이 장이 끝나면

이 장을 마치면 read_single_line 과 emit_sum 을 결합한 실행 파일 하나를 여섯 경우(정상·CRLF·나쁜 토큰·버퍼 초과 긴 줄·둘째 줄 존재·쓰기 실패)로 돌린다. 어느 실패에서도 표준 출력은 정확히 0바이트이고, 표준 오류와 종료 상태 1 만 남는 것을 눈으로 확인한다.

이 장에서 답할 질문

  1. fgets 버퍼에 100바이트 줄이 들어왔는데 LINE_CAP 이 64 면 뒷부분은 읽지 않고 그대로 끝내도 괜찮은가?
  2. LINE_CAP 을 main.c 에서 4096 처럼 고정 크기로 잡으면 여전히 너무 긴 줄에서 막히지 않는가?

103.1 읽기 경계 — EOF · 읽기 실패 · 긴 줄 · 둘째 줄#

다음 시연은 앞 두 장의 파서와 ckd_add 합산 위에 read_single_line 과 emit_sum 을 얹어 실제 프로세스 입출력으로 검증한다. 긴 줄 경계를 작은 입력으로 시험하려고 줄 버퍼 용량을 LINE_CAP = 64 로 잡았다.

examples/sumcsv/04-io/sumcsv_io.c

/* 긴 줄 잘림, 둘째 줄, 읽기 실패, 쓰기/닫기 실패를 모두 잡고 오류 시 stdout 을 비운다. */
#include <ctype.h>
#include <errno.h>
#include <inttypes.h>
#include <limits.h>
#include <stdckdint.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

enum { LINE_CAP = 64 };

static const char *skip_ws(const char *p)
{
    while (*p == ' ' || *p == '\t') p++;
    return p;
}

static bool at_eol(const char *p)
{
    return *p == '\0' || *p == '\n' || (*p == '\r' && p[1] == '\n');
}

static bool parse_i64_field(const char *start, const char **out_rest,
                            int64_t *out_val, const char **out_why)
{
    const char *p = skip_ws(start);
    if (at_eol(p) || *p == ',') {
        *out_why = "empty field";
        return false;
    }
    const char *digits = p;
    if (*digits == '+' || *digits == '-') digits++;
    if (!isdigit((unsigned char)*digits)) {
        *out_why = "bad integer";
        return false;
    }

    errno = 0;
    char *end = nullptr;
    long long v = strtoll(p, &end, 10);
    if (end == p) {
        *out_why = "bad integer";
        return false;
    }
    if (errno == ERANGE) {
        *out_why = "integer out of range";
        return false;
    }
#if LLONG_MIN < INT64_MIN || LLONG_MAX > INT64_MAX
    if (v < INT64_MIN || v > INT64_MAX) {
        *out_why = "integer out of range";
        return false;
    }
#endif

    *out_val = (int64_t)v;
    *out_rest = end;
    return true;
}

static bool sum_line(const char *line, int64_t *out_sum, const char **out_why)
{
    const char *p = skip_ws(line);
    if (at_eol(p)) {
        *out_why = "empty input";
        return false;
    }

    int64_t acc = 0;
    for (;;) {
        int64_t val = 0;
        const char *rest = nullptr;
        if (!parse_i64_field(p, &rest, &val, out_why)) return false;
        if (ckd_add(&acc, acc, val)) {
            *out_why = "sum overflow";
            return false;
        }
        p = skip_ws(rest);
        if (*p == ',') {
            p++;
            if (at_eol(skip_ws(p))) {
                *out_why = "trailing comma";
                return false;
            }
            continue;
        }
        if (*p == '\r' && p[1] == '\n') p += 2;
        else if (*p == '\n') p++;
        if (*p == '\0') {
            *out_sum = acc;
            return true;
        }
        *out_why = "unexpected character";
        return false;
    }
}

static bool read_single_line(FILE *in, char *buf, size_t cap, const char **out_why)
{
    if (!fgets(buf, (int)cap, in)) {
        *out_why = ferror(in) ? "read error" : "empty input";
        return false;
    }
    size_t len = strlen(buf);
    if (len > 0 && buf[len - 1] == '\n') {
        int extra = fgetc(in);
        if (ferror(in)) {
            *out_why = "read error";
            return false;
        }
        if (extra != EOF) {
            *out_why = "extra line";
            return false;
        }
        return true;
    }
    if (len == cap - 1) {
        int next = fgetc(in);
        if (ferror(in)) {
            *out_why = "read error";
            return false;
        }
        if (next != EOF) {
            *out_why = "line too long";
            return false;
        }
    }
    return true;
}

static bool emit_sum(FILE *out, int64_t sum, const char **out_why)
{
    if (fprintf(out, "%" PRId64 "\n", sum) < 0 || fflush(out) != 0 || ferror(out)) {
        *out_why = "write error";
        return false;
    }
    return true;
}

int main(int argc, char **argv)
{
    bool test_write_fail = (argc == 2 && strcmp(argv[1], "--test-write-fail") == 0);
    if (argc > 1 && !test_write_fail) {
        fprintf(stderr, "sumcsv: unexpected argument\n");
        return 1;
    }

    char line[LINE_CAP];
    const char *why = "";
    if (!read_single_line(stdin, line, sizeof line, &why)) {
        fprintf(stderr, "sumcsv: %s\n", why);
        return 1;
    }

    int64_t sum = 0;
    if (!sum_line(line, &sum, &why)) {
        fprintf(stderr, "sumcsv: %s\n", why);
        return 1;
    }

    if (test_write_fail) {
        FILE *ro = fopen("/dev/null", "r");
        if (!ro) {
            fprintf(stderr, "sumcsv: write error\n");
            return 1;
        }
        bool ok = emit_sum(ro, sum, &why);
        fclose(ro);
        if (!ok) {
            fprintf(stderr, "sumcsv: %s\n", why);
            return 1;
        }
        return 0;
    }

    if (!emit_sum(stdout, sum, &why)) {
        fprintf(stderr, "sumcsv: %s\n", why);
        return 1;
    }
    return 0;
}

실행 결과

  basic            rc=0 out_bytes=3  stdout="60" stderr=""
  crlf             rc=0 out_bytes=2  stdout="7" stderr=""
  bad-token        rc=1 out_bytes=0  stdout="" stderr="sumcsv: bad integer"
  overlong         rc=1 out_bytes=0  stdout="" stderr="sumcsv: line too long"
  two-lines        rc=1 out_bytes=0  stdout="" stderr="sumcsv: extra line"
  write-fail       rc=1 out_bytes=0  stdout="" stderr="sumcsv: write error"

여섯 줄의 실행 결과가 여섯 가지 I/O 경로를 전부 보여 준다.

read_single_line 이 버퍼 크기에 딱 맞춘 경우와 진짜 넘친 경우를 어떻게 가르는지 살펴본다. fgets(buf, cap, in) 은 최대 cap - 1 바이트를 읽고 NUL 을 붙인다.

  • buf[len - 1] == '\n' 이면 한 줄을 끝까지 다 읽은 것이다. 여기서 멈추지 않고 fgetc(in) 을 한 번 더 불러 ferror 가 아닌지 보고, EOF 가 아니라면 둘째 줄이 남아 있으므로 "extra line" 으로 거부한다.
  • 끝 글자가 '\n' 이 아니고 len == cap - 1 이면 갈림길이다 — 개행 없는 마지막 줄이 우연히 정확히 cap - 1 글자였을 수도 있고, 더 긴 줄이 잘렸을 수도 있다. 이때도 fgetc(in) 한 글자를 엿보아 EOF 면 통과시키고, 한 글자라도 더 있으면 "line too long" 으로 거부한다.

★ 엿보기 한 번(fgetc)이 잘림과 딱 맞음을 가른다. 이 한 줄이 없으면 63바이트짜리 유효한 마지막 줄을 잘못 막거나, 64바이트를 넘긴 줄을 끝에서 잘라 합산해 버린다.

103.2 쓰기 경계와 실패 시 stdout 0바이트#

78장 에서 보았듯 fprintf 가 양수를 돌려주었더라도 실제로는 사용자 공간의 스트림 버퍼에만 들어갔을 뿐 커널로 나가지 않았을 수 있다. 그래서 emit_sum 은 fprintf < 0 만 보지 않고 fflush(out) != 0 || ferror(out) 까지 한꺼번에 확인한다.

시연의 --test-write-fail 경로가 이 검사가 실제로 무는지 확인하는 장치다. /dev/null 을 읽기 전용("r")으로 연 스트림에 emit_sum 을 부르면 fprintf 나 fflush 단계에서 오류가 걸리고, 프로세스는 stderr 에 "sumcsv: write error" 를 남긴 뒤 종료 1 로 끝난다. 실제 파이프가 깨지거나 디스크가 꽉 찬 상황에서도 같은 경로가 작동한다.

흔한 오해. “표준 출력에 값이 나갔더라도 종료 코드 1 만 주면 호출자가 오류인 줄 안다”

많은 파이프라인과 셸 치환(out=$(./sumcsv < in.txt))은 먼저 표준 출력을 모으거나 다음 필터로 넘기고, 심지어 sumcsv 가 낸 줄을 숫자로 믿고 다음 계산에 쓴다. 출력과 종료 상태를 따로 검증하지 않는 호출자 앞에서 부분 출력이나 오류 안내 문구를 stdout 에 섞으면 장애가 번진다.

그래서 1판 계약은 오류 시 stdout 0바이트를 못박는다. 읽고·해석하고·더하는 일이 모두 성공한 뒤에만 emit_sum 이 stdout 을 연다.

문. fgets 버퍼에 100바이트 줄이 들어왔는데 LINE_CAP 이 64 면 뒷부분은 읽지 않고 그대로 끝내도 괜찮은가?

답. 지금처럼 한 번 실행에 한 줄만 읽고 곧장 종료 1 로 끝나는 명령줄 필터에서는 안전하다. 반면 같은 FILE * 에서 다음 레코드를 계속 읽는 서버나 반복문 안에서는 남은 \n 까지 마저 읽어 버리지 않으면 다음 호출이 아까 줄의 뒷토막을 새 줄로 오해한다. 한 번 돌고 끝나는지, 스트림을 이어 쓰는지 따라 소비 전략을 고른다.

문. LINE_CAP 을 main.c 에서 4096 처럼 고정 크기로 잡으면 여전히 너무 긴 줄에서 막히지 않는가?

답. 그렇다. 고정 버퍼를 쓰는 1판에서는 4,095바이트를 넘는 한 줄이 조용히 잘리지 않고 명시적으로 거부된다는 점이 중요하다 — 예측 가능한 한도와 정직한 거부다. 임의 길이 줄을 모두 받아야 하는 요구로 자라면 힙 위 동적 버퍼가 필요하고, 그 구조와 비용을 105장 에서 직접 짠다.

복습 정리

이로써 1판 sumcsv 를 이루는 세 기둥 — 파서(101장), 검사 산술(102장), 입출력 경계(이 장) — 이 모두 한 흐름으로 맞물렸다. 다음 장(104장)에서는 한 파일에 담겨 있던 코드를 인터페이스와 구현 파일로 나누고, Makefile 과 골든 시험과 자동 시험으로 굳힌다.
