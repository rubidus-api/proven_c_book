돌아보기

앞의 다섯 장에서 표준 C23 만으로 오류 경로와 골든 시험까지 갖춘 sumcsv 1판을 완성했다. 그러면 1판 위에 「여러 줄을 읽고, 사용자가 고른 열 하나만 더하며, 쉼표나 따옴표가 든 따옴표 필드도 받고, 줄 길이 상한도 없애 달라」는 요구를 얹으면 어떤 변화가 일어나는가?

답. 단순히 if 문 몇 줄이 늘어나는 것이 아니다. 길이를 미리 알 수 없는 줄과 따옴표 벗긴 셀 두 곳을 동시에 담을 동적 버퍼 둘이 필요해지고, 버퍼마다 곱셈 넘침 검사와 realloc 실패 보존이 따라붙으며, 함수가 도중에 실패할 때 두 버퍼를 빠짐없이 free 하도록 정리 사다리(goto cleanup) 를 세워야 한다.

★ 순수 C 로 끝까지 짤 수는 있지만, 소유권·복사·정리·오류 전달·기억 출처 다섯 가지를 매번 손으로 챙기는 마찰이 코드 절반을 차지하기 시작한다.

이 장의 필요성과 맥락

이 장은 제13부의 마무리이자 다음 부(106장 이후)로 건너가는 다리다. 남이 만든 라이브러리를 소개하기 전에, 표준 C 만으로 실제 CSV 요구 확장(sumcsv 2판)을 끝까지 짜 보아 어디까지는 순수 C 로 깨끗이 풀리고, 어디서부터 사람이 매번 반복해 챙겨야 하는 부담이 커지는지 손으로 재어 보기 위해서다.

이 장이 끝나면

이 장을 마치면 여러 줄·열 선택·따옴표 필드("" 이스케이프)·가변 줄 버퍼·NUL 바이트 차단까지 구현한 순수 C 2판 sumcsv_v2.c 를 직접 실행해 확인한다. 이어 1판과 2판 사이의 비용 증가를 표로 견주어, 다음 제14부가 도입하는 API 계약이 정확히 어떤 마찰을 덜어 주려는지 분명히 본다.

이 장에서 답할 질문

  1. sumcsv_v2.c 에서 따옴표 안에 줄바꿈("\n")이 들어 있는 CSV 도 통과하는가?
  2. dynbuf 가 이미 잘 도는데, 왜 기억 출처(할당자 고정)를 아쉬운 점으로 적었는가?

105.1 2판이 요구하는 네 가지와 순수 C 구현#

다음 시연은 1판에 네 가지 새 요구 — ① 여러 줄 스트림 합산, ② 지정한 열(target_col) 하나만 더하기, ③ 큰따옴표로 감싼 필드와 "" 이스케이프 처리, ④ 길이 제한 없는 동적 줄·셀 버퍼 — 를 모두 표준 C23 만으로 구현한 판이다.

examples/sumcsv/06-next/sumcsv_v2.c

/* 여러 줄 · 열 선택 · 따옴표 필드 · 동적 줄 버퍼까지 손수 구현한 2판 ---
   요구가 늘수록 기억 수명과 오류 정리가 C 코드에 어떻게 쌓이는지 보인다. */
#include <ctype.h>
#include <errno.h>
#include <inttypes.h>
#include <limits.h>
#include <stdckdint.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct dynbuf {
    char  *data;
    size_t len;
    size_t cap;
};

static bool buf_push(struct dynbuf *b, char ch)
{
    if (b->len + 1 >= b->cap) {
        size_t ncap = 0;
        if (b->cap == 0) {
            ncap = 32;
        } else if (ckd_mul(&ncap, b->cap, (size_t)2)) {
            return false;
        }
        char *nb = realloc(b->data, ncap);
        if (!nb) return false;
        b->data = nb;
        b->cap = ncap;
    }
    b->data[b->len++] = ch;
    b->data[b->len] = '\0';
    return true;
}

static int read_dyn_line(FILE *in, struct dynbuf *b, const char **why)
{
    b->len = 0;
    if (b->data) b->data[0] = '\0';

    int ch = fgetc(in);
    if (ch == EOF) {
        if (ferror(in)) { *why = "read error"; return -1; }
        return 0;
    }
    do {
        if (ch == '\0') { *why = "embedded NUL byte"; return -1; }
        if (!buf_push(b, (char)ch)) { *why = "out of memory"; return -1; }
        if (ch == '\n') break;
        ch = fgetc(in);
    } while (ch != EOF);

    if (ferror(in)) { *why = "read error"; return -1; }
    return 1;
}

static bool parse_i64_exact(const char *s, int64_t *out, const char **why)
{
    while (*s == ' ' || *s == '\t') s++;
    if (*s == '\0') { *why = "empty cell"; return false; }

    const char *digits = s;
    if (*digits == '+' || *digits == '-') digits++;
    if (!isdigit((unsigned char)*digits)) { *why = "bad integer"; return false; }

    errno = 0;
    char *end = nullptr;
    long long v = strtoll(s, &end, 10);
    if (end == s) { *why = "bad integer"; return false; }
    if (errno == ERANGE) { *why = "integer out of range"; return false; }
#if LLONG_MIN < INT64_MIN || LLONG_MAX > INT64_MAX
    if (v < INT64_MIN || v > INT64_MAX) { *why = "integer out of range"; return false; }
#endif
    while (*end == ' ' || *end == '\t') end++;
    if (*end != '\0') { *why = "trailing garbage"; return false; }
    *out = (int64_t)v;
    return true;
}

static bool eval_row_col(const char *line, size_t target_col, struct dynbuf *cell,
                         int64_t *out_val, const char **why)
{
    const char *p = line;
    size_t col = 0;
    bool found = false;

    for (;;) {
        col++;
        cell->len = 0;
        if (cell->data) cell->data[0] = '\0';

        if (*p == '"') {
            p++;
            for (;;) {
                if (*p == '\0' || *p == '\n' || (*p == '\r' && p[1] == '\n')) {
                    *why = "unclosed quote";
                    return false;
                }
                if (*p == '"') {
                    if (p[1] == '"') {
                        if (!buf_push(cell, '"')) { *why = "out of memory"; return false; }
                        p += 2;
                        continue;
                    }
                    p++;
                    break;
                }
                if (!buf_push(cell, *p)) { *why = "out of memory"; return false; }
                p++;
            }
            while (*p == ' ' || *p == '\t') p++;
        } else {
            while (*p && *p != ',' && *p != '\n' && !(*p == '\r' && p[1] == '\n')) {
                if (*p == '"') { *why = "stray quote"; return false; }
                if (!buf_push(cell, *p)) { *why = "out of memory"; return false; }
                p++;
            }
        }

        if (col == target_col) {
            if (!parse_i64_exact(cell->data ? cell->data : "", out_val, why)) return false;
            found = true;
        }

        if (*p == ',') { p++; continue; }
        if (*p == '\r' && p[1] == '\n') p += 2;
        else if (*p == '\n') p++;
        if (*p != '\0') { *why = "unexpected char after field"; return false; }
        break;
    }

    if (!found) {
        *why = "missing target column";
        return false;
    }
    return true;
}

static bool sum_csv_stream(FILE *in, size_t target_col, int64_t *out_sum, const char **out_why)
{
    struct dynbuf line = {0};
    struct dynbuf cell = {0};
    int64_t acc = 0;
    size_t rows = 0;
    bool ok = false;

    for (;;) {
        int rc = read_dyn_line(in, &line, out_why);
        if (rc < 0) goto cleanup;
        if (rc == 0) break;

        rows++;
        int64_t v = 0;
        if (!eval_row_col(line.data, target_col, &cell, &v, out_why)) goto cleanup;
        if (ckd_add(&acc, acc, v)) { *out_why = "sum overflow"; goto cleanup; }
    }

    if (rows == 0) {
        *out_why = "empty input";
        goto cleanup;
    }
    *out_sum = acc;
    ok = true;

cleanup:
    free(line.data);
    free(cell.data);
    return ok;
}

static void run_case(const char *label, size_t col, const char *csv)
{
    FILE *fp = tmpfile();
    if (!fp) {
        printf("  %-18s -> err: tmpfile failed\n", label);
        return;
    }
    size_t n = strlen(csv);
    if (n > 0 && fwrite(csv, 1, n, fp) != n) {
        fclose(fp);
        printf("  %-18s -> err: write failed\n", label);
        return;
    }
    rewind(fp);

    int64_t sum = -1;
    const char *why = "";
    if (sum_csv_stream(fp, col, &sum, &why)) {
        printf("  %-18s col=%zu -> ok: %" PRId64 "\n", label, col, sum);
    } else {
        printf("  %-18s col=%zu -> err: %s (sum untouched: %" PRId64 ")\n",
               label, col, why, sum);
    }
    fclose(fp);
}

int main(void)
{
    run_case("quoted multiline", 2,
             "\"widget, small\",100\n\"bolt \"\"hex\"\"\",250\nplain,-30\r\n");
    run_case("unclosed quote",   2, "\"widget,100\n");
    run_case("missing col",      3, "a,10\n");
    run_case("bad target cell",  2, "\"ok,name\",abc\n");
    return 0;
}

실행 결과

  quoted multiline   col=2 -> ok: 320
  unclosed quote     col=2 -> err: unclosed quote (sum untouched: -1)
  missing col        col=3 -> err: missing target column (sum untouched: -1)
  bad target cell    col=2 -> err: bad integer (sum untouched: -1)

코드가 208줄로 불어났고, 네 가지 안전장치가 새로 들어왔다.

  • ckd_mul 로 용량을 두 배씩 키우는 struct dynbuf — buf_push 는 b->cap * 2 에서 size_t 가 넘치지 않도록 ckd_mul 로 먼저 묻고, realloc 이 널을 돌려주면 기존 b->data 를 덮어쓰지 않은 채 false 로 돌아간다.
  • 내장 NUL 바이트 차단 — CSV 입력 한가운데 \0 바이트가 섞이면 C 문자열 함수들이 그 자리에서 끝났다고 오판하므로, read_dyn_line 이 fgetc 단계에서 ch == '\0' 을 "embedded NUL byte" 로 거부한다.
  • 따옴표 상태와 "" 이스케이프 — 여는 " 로 시작한 필드는 안의 , 를 구분자로 치지 않고, "" 연속 두 개는 리터럴 " 하나로 펼쳐 cell 버퍼에 옮겨 담는다. 닫히지 않은 따옴표(unclosed quote)나 따옴표 바깥에 끼어든 따옴표(stray quote)는 그 자리에서 거부한다.
  • goto cleanup 과 출력 보존 — sum_csv_stream 은 line 과 cell 두 동적 버퍼를 열기 때문에, 읽기 실패·파싱 실패·빈 입력·합산 넘침 어느 갈래에서 빠져나오더라도 cleanup: 라벨 한곳을 거쳐 두 버퍼를 모두 free 하고, 호출자 sum = -1 은 성공 시에만 덮어쓴다.

105.2 1판에서 2판으로 옮기며 쌓인 마찰#

1판의 다섯 단계가 문제 없었는데도, 2판 한 편을 짜는 동안 순수 C 에 반복적으로 들어간 몫을 나란히 적어 본다.

★ 어느 한 줄도 언어 오류가 아니며, 실제로 새니타이저와 -Werror 아래서 깨끗이 통과했다. 다만 문제는 규모다. 지금은 dynbuf 가 둘뿐이지만 필드를 구조체 배열로 모아 반환하거나 중첩 구조를 읽기 시작하면 buf_push·cleanup: 사다리· 인자별 할당 계약이 파일마다 되풀이된다. 사람이 한 자리라도 빠뜨리면 누수나 잘림이나 사용 후 해제가 된다.

흔한 오해. “순수 C 로도 다 짤 수 있는데, 굳이 제14부 같은 계약과 라이브러리를 볼 필요가 있는가”

짤 수 있다는 사실 자체가 맞다 — 방금 208줄짜리 sumcsv_v2.c 로 그것을 증명했다. 제14부는 “표준 C 로 안 된다”가 아니라 매번 손으로 챙기던 다섯 약속(오류 값, 길이·뷰, 할당자 매개변수, 실패 후 상태, 거부 정책)을 타입과 시그니처 안으로 옮기면 코드가 어떻게 달라지는가를 묻는 자리다. 손수 짜 본 사람이어야 그 대조의 무게를 안다.

문. sumcsv_v2.c 에서 따옴표 안에 줄바꿈("\n")이 들어 있는 CSV 도 통과하는가?

답. 아니다. 2판 예제는 read_dyn_line 이 \n 에서 한 줄을 끊으므로, 따옴표 안의 실제 개행 문자가 오면 "unclosed quote" 로 거부한다. RFC 4180 식 여러 줄 따옴표 셀까지 받으려면 줄 단위 읽기 자체가 따옴표 상태를 알아야 하고, 스트림 입력 전체를 다루는 상태 기계로 한 번 더 커진다.

문. dynbuf 가 이미 잘 도는데, 왜 기억 출처(할당자 고정)를 아쉬운 점으로 적었는가?

답. 명령줄 도구 한 번 돌리고 끝나는 프로세스에서는 힙 realloc/free 로 충분하다. 그러나 임베디드(99장)처럼 힙을 못 쓰는 자리나 요청마다 통째로 비우는 서버 아레나 안에서는 sum_csv_stream 이 내부에서 realloc 을 부른다는 사실 하나 때문에 함수 전체를 재사용하지 못한다.

복습 정리

이 장이 제13부의 마지막 장이다. 앞의 백다섯 장에서 배운 표준 C 만으로 작은 도구 하나를 명세(100장)부터 다중 파일 빌드와 시험, 그리고 2판 확장까지 손수 지어 보았다. 다음 제14부는 106장 에서 C 가 반세기 동안 같은 자리에서 미끄러져 온 다섯 버그를 짚은 뒤, 이 부에서 손으로 챙겼던 규율을 라이브러리 설계로 옮기는 길을 따라가고, 116장 에서 다시 한번 순수 C 와 라이브러리 판을 나란히 세워 비교한다.
